# Fallback profiles are explicitly synthetic screening envelopes, not field measurements.
# The 20 profiles above remain the existing curated starter profiles. For other crops,
# a crop-family envelope is varied deterministically by crop name so all 100 classes
# have distinct feature distributions without overwriting any real observations.
def get_crop_profile(crop):
    if crop in CROP_PROFILES:
        profile = dict(CROP_PROFILES[crop])
        profile.setdefault("humidity", 72 if profile.get("water") == "High" else 62)
        return profile, True

    if crop in CROP_GROUPS["water_high"]:
        family, water, temp, rain, humidity = "water", "High", 27, 1150, 78
        soils = ["Alluvial", "Loamy", "Black cotton soil"]
    elif crop in CROP_GROUPS["dry"]:
        family, water, temp, rain, humidity = "dry", "Low", 27, 560, 48
        soils = ["Red soil", "Sandy", "Loamy"]
    elif crop in CROP_GROUPS["cool"]:
        family, water, temp, rain, humidity = "cool", "Moderate", 19, 650, 60
        soils = ["Loamy", "Alluvial", "Sandy"]
    elif crop in CROP_GROUPS["hort"]:
        family, water, temp, rain, humidity = "hort", "Moderate", 25, 820, 68
        soils = ["Loamy", "Red soil", "Alluvial"]
    else:
        family, water, temp, rain, humidity = "general", "Moderate", 25, 780, 64
        soils = ["Loamy", "Red soil", "Laterite"]

    stable_seed = sum((index + 1) * ord(char) for index, char in enumerate(crop))
    profile = {
        "soils": soils,
        "ph": round(5.7 + (stable_seed % 17) * 0.085, 2),
        "temp": round(temp + ((stable_seed % 9) - 4) * 0.55, 1),
        "rain": round(rain * (0.82 + (stable_seed % 37) / 100), 1),
        "humidity": int(np.clip(humidity + (stable_seed % 19) - 9, 35, 92)),
        "water": water,
        "n": int(np.clip(38 + stable_seed % 83, 20, 130)),
        "p": int(np.clip(24 + (stable_seed // 7) % 47, 15, 90)),
        "k": int(np.clip(26 + (stable_seed // 13) % 66, 15, 110))
    }
    return profile, False

def estimate_real_yield_tonnes_per_acre(crop, selected_state, selected_district):
    """Return the fixed crop benchmark converted from kg/ha to tonnes/acre."""
    benchmark_kg_per_hectare = STATE_CROP_YIELD_BENCHMARKS_KG_HA.get(
        (selected_state, crop), CROP_YIELD_BENCHMARK_KG_HA.get(crop)

def best_match(value, choices, field_name="value", threshold=70):

    # Normalize both the user's entry and database choices so matching is
    # case-insensitive and tolerant of extra spaces and spelling mistakes.
    normalized_choices = [
        " ".join(str(choice).casefold().split())
        for choice in choices
    ]

    raw_value = value

    while True:
        normalized_value = " ".join(str(raw_value).casefold().split())

        match = process.extractOne(
            normalized_value,
            normalized_choices,
            scorer=fuzz.WRatio
        )

        if match and match[1] >= threshold:
            return choices[match[2]]

        raw_value = input(
            f"Invalid {field_name}. Please re-enter: "
        ).strip()


def ask_float(prompt, low, high, suffix=""):

    while True:

        raw = input(
            f"{prompt} [{low}-{high}]{suffix}: "
        ).strip()

        try:
            value = float(raw)

            if low <= value <= high:
                return value

        except ValueError:
            pass

        if prompt == "Average temperature":
            print("Enter a number between 0 and 50.")
        else:
            print(
                f"Please enter a number between {low} and {high}."
            )


def ask_choice(prompt, choices):

    print("\n" + prompt)

    print(" | ".join(choices))

    raw = input("Your choice: ").strip()

    field_name = prompt.replace("Select ", "").rstrip(":").lower()
    return best_match(raw, choices, field_name)


def haversine_km(lat1, lon1, lat2, lon2):

    r = 6371

    p1 = math.radians(lat1)
    p2 = math.radians(lat2)

    dp = math.radians(lat2 - lat1)
    dl = math.radians(lon2 - lon1)

    a = (
        math.sin(dp / 2) ** 2
        + math.cos(p1) * math.cos(p2)
        * math.sin(dl / 2) ** 2
    )

    return 2 * r * math.asin(math.sqrt(a))

        normalized_choice = " ".join(crop_choice.casefold().split())
        normalized_top_names = [name.casefold() for name in top_crop_names]
        crop_match = max(
            (process.extractOne(normalized_choice, normalized_top_names, scorer=scorer)
             for scorer in (fuzz.WRatio, fuzz.partial_ratio, fuzz.ratio)),
            key=lambda match: match[1] if match else -1
        )

        if crop_match:
            selected_crop = top_crop_names[crop_match[2]]

            elif factor_name == "Temperature":

                    f"Temperature: entered {actual_value:.1f} {unit}, "

    # its rainfall requirement when comparing the available sowing months.
    crop_average_rainfall = float(
        dataset_crop_profiles[selected_crop]["rainfall_mm"]

    monthly_district_rainfall = (
        district_rain.groupby("month")["rainfall_mm"].mean().to_dict()

    def expected_seasonal_rainfall(sowing_month, crop_duration_days):
        month_days = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]
        remaining_days = int(crop_duration_days)
        month = int(sowing_month)
        total_rainfall = 0.0

        while remaining_days > 0:
            days_in_month = month_days[month - 1]
            days_in_period = min(remaining_days, days_in_month)
            monthly_average = float(
                monthly_district_rainfall.get(
                    month, district_rain["rainfall_mm"].mean()
                )
            )
            total_rainfall += monthly_average * days_in_period / days_in_month
            remaining_days -= days_in_period
            month = month % 12 + 1

        return total_rainfall

    crop_rows["expected_seasonal_rainfall_mm"] = crop_rows.apply(
        lambda row: expected_seasonal_rainfall(

    def concise_guideline(value, section):
        """Format crop-specific advice as short, readable field instructions."""
        value = re.sub(r";\s*", ". ", str(value).strip())
        value = re.sub(r"[ \t]+", " ", value)

        if section == "ipm_guidance":
            formatted = []
            for line in value.splitlines():
                line = line.strip()
                if not line:
                    continue
                match = re.match(r"^(\d+\.\s*)(.*)$", line)
                if match:
                    prefix, content = match.groups()
                    wrapped = textwrap.fill(
                        content, width=84,
                        initial_indent="  " + prefix,
                        subsequent_indent="     "
                    )
                else:
                    wrapped = textwrap.fill(
                        line, width=88, initial_indent="  ",
                        subsequent_indent="  "
                    )
                formatted.append(wrapped)
            return "\n\n".join(formatted)

        if section in {"common_diseases", "common_pests"}:
            formatted = []
            for item in value.splitlines():
                item = item.strip().lstrip("• ")
                if not item:
                    continue
                if item.startswith("CROP-SPECIFIC "):
                    formatted.append(item)
                    continue
                if item.startswith("PREVENTION:"):
                    formatted.append(textwrap.fill(
                        item, width=88, initial_indent="  ",
                        subsequent_indent="  "
                    ))
                    continue
                formatted.append(textwrap.fill(
                    item, width=88, initial_indent="  • ",
                    subsequent_indent="    "
                ))
            return "\n\n".join(formatted)

        paragraphs = [part.strip() for part in re.split(r"\n+", value) if part.strip()]
        return "\n\n".join(
            textwrap.fill(part, width=88, subsequent_indent="  ")
            for part in paragraphs
        )

    def predict_market_price_for_date(target_market, target_district, target_state, target_date):
        """Fit LinearRegression to prior real or synthetic observations for this date.

        history only when a narrower history is too short. Synthetic observations
        remain labelled as estimates and are never represented as real mandi quotes.

            ("state", crop_market_data[crop_market_data["State"].astype(str).str.casefold() == str(target_state).casefold()])
        ]

        forecast_cutoff = pd.Timestamp(target_date)

                scope_rows["Arrival_Date"] < forecast_cutoff

                if chosen_history is None:

                    chosen_scope = scope_name
                if candidate_history["Arrival_Date"].nunique() >= 3:

                        candidate_history["arrival_date_ordinal"].to_numpy(dtype=float),
                        candidate_history["arrival_month_sin"].to_numpy(dtype=float),
                        candidate_history["arrival_month_cos"].to_numpy(dtype=float)
                    ])
                    y_values = candidate_history["Modal_x0020_Price"].to_numpy(dtype=float)
                    trend_model = LinearRegression()
                    trend_model.fit(x_values, y_values)
                    predicted = float(trend_model.predict(market_date_features(target_date))[0])
                    if scope_name == "market" and "data_source" in candidate_history.columns:
                        if candidate_history["data_source"].astype(str).str.startswith("synthetic").all():
                            return max(0.01, predicted), "linear_regression_synthetic_estimate"
                    return max(0.01, predicted), f"linear_regression_{scope_name}"

        # A one-day snapshot has no trend. Use only the latest observed real quote
        # from the narrowest available market/district/state scope as a reference.
        if chosen_history is None:
            return np.nan, "unavailable"
        latest_date = chosen_history["Arrival_Date"].max()

    grading_cost = GRADING_COST_PER_QUINTAL * quantity_quintals
    market_charge = MARKET_CHARGE_PER_QUINTAL * quantity_quintals

    # Dynamic post-harvest exposure includes time waiting after harvest,
    # handling time and estimated road transit to the selected actual mandi.
    # State-anchor distance is used because the current location catalogue
    # does not contain precise farm coordinates.
    market_distance_lookup = (
        nearby.drop_duplicates(["market", "district", "state"])
        .set_index(["market", "district", "state"])["distance_km"].to_dict()
    )
    assumed_transport_speed_kmph = 35.0
    handling_buffer_hours = 2.0
    crop_shelf_life_days = max(1.0, float(CROP_SHELF_LIFE_DAYS.get(selected_crop, 30)))
    options_df["market_distance_km"] = options_df.apply(
        lambda row: float(market_distance_lookup.get(
            (row["market"], row["district"], row["state"]), 0.0
        )),
        axis=1
    )
    options_df["estimated_transit_hours"] = (
        options_df["market_distance_km"] / assumed_transport_speed_kmph
    )
    options_df["estimated_waiting_hours"] = (
        options_df["days_after_harvest"] * 24.0 + handling_buffer_hours
    )
    options_df["total_postharvest_age_days"] = (
        options_df["estimated_waiting_hours"]
        + options_df["estimated_transit_hours"]
    ) / 24.0
    options_df["spoilage_penalty_fraction"] = np.clip(
        options_df["total_postharvest_age_days"] / crop_shelf_life_days,
        0.0, 0.80
    )

    # Distance-based transportation costs use fixed rates, never selling-price percentages.
    options_df["transportation_cost"] = (
        TRANSPORT_COST_PER_QUINTAL_KM
        * quantity_quintals
        * options_df["market_distance_km"]
    )
    options_df["total_operating_expenses"] = (
        field_labour_cost
        + harvesting_labour_cost
        + total_seed_cost
        + packaging_cost
        + grading_cost
        + options_df["transportation_cost"]
        + market_charge
    )

    # Gross revenue is calculated before the separate perishability deduction.
    options_df["gross_revenue"] = options_df["expected_price"] * quantity_quintals
    options_df["perishability_loss_amount"] = (
        options_df["gross_revenue"] * options_df["spoilage_penalty_fraction"]
    )
    options_df["combined_loss_expense_rate"] = np.where(
        options_df["gross_revenue"] > 0,
        options_df["total_operating_expenses"] / options_df["gross_revenue"] * 100.0,
        0.0
    )
    options_df["estimated_revenue"] = options_df["gross_revenue"]
    options_df["estimated_profit"] = (
        options_df["gross_revenue"]
        - options_df["perishability_loss_amount"]
        - options_df["total_operating_expenses"]
    )

    best_option = options_df.sort_values(
        ["estimated_profit", "days_after_harvest"],
        ascending=[False, True]
    ).iloc[0]

        options_df.days_after_harvest == 0

    )

        f"| {'Grading cost:':<27} ₹{grading_cost:>14,.0f} |",
