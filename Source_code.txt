# Fallback profiles are explicitly synthetic screening envelopes, not field measurements.
# The 20 profiles above remain the existing curated starter profiles. For other crops,
# a crop-family envelope is varied deterministically by crop name so all 100 classes
# have distinct feature distributions without overwriting any real observations.
def get_crop_profile(crop):
    if crop in CROP_PROFILES:
        profile = dict(CROP_PROFILES[crop])
        profile.setdefault("humidity", 72 if profile.get("water") == "High" else 62)
        return profile, True

    if crop in CROP_GROUPS["water_high"]:
        family, water, temp, rain, humidity = "water", "High", 27, 1150, 78
        soils = ["Alluvial", "Loamy", "Black cotton soil"]
    elif crop in CROP_GROUPS["dry"]:
        family, water, temp, rain, humidity = "dry", "Low", 27, 560, 48
        soils = ["Red soil", "Sandy", "Loamy"]
    elif crop in CROP_GROUPS["cool"]:
        family, water, temp, rain, humidity = "cool", "Moderate", 19, 650, 60
        soils = ["Loamy", "Alluvial", "Sandy"]
    elif crop in CROP_GROUPS["hort"]:
        family, water, temp, rain, humidity = "hort", "Moderate", 25, 820, 68
        soils = ["Loamy", "Red soil", "Alluvial"]
    else:
        family, water, temp, rain, humidity = "general", "Moderate", 25, 780, 64
        soils = ["Loamy", "Red soil", "Laterite"]

    stable_seed = sum((index + 1) * ord(char) for index, char in enumerate(crop))
    profile = {
        "soils": soils,
        "ph": round(5.7 + (stable_seed % 17) * 0.085, 2),
        "temp": round(temp + ((stable_seed % 9) - 4) * 0.55, 1),
        "rain": round(rain * (0.82 + (stable_seed % 37) / 100), 1),
        "humidity": int(np.clip(humidity + (stable_seed % 19) - 9, 35, 92)),
        "water": water,
        "n": int(np.clip(38 + stable_seed % 83, 20, 130)),
        "p": int(np.clip(24 + (stable_seed // 7) % 47, 15, 90)),
        "k": int(np.clip(26 + (stable_seed // 13) % 66, 15, 110))
    }
    return profile, False

def estimate_real_yield_tonnes_per_acre(crop, selected_state, selected_district):

    # Prefer the optional wide-format real dataset when it contains a district-level
    # yield explicitly measured in tonnes/hectare. Nuts and bales are never converted
    # to tonnes without a crop-specific physical conversion factor.
    if not wide_yield_df.empty:
        wide_crop_rows = wide_yield_df[
            (wide_yield_df["crop_name"].astype(str).str.casefold() == crop.casefold())
            & wide_yield_df["yield_tonnes_per_hectare"].notna()
            & (wide_yield_df["yield_tonnes_per_hectare"] > 0)
        ].copy()
        normalize_location = lambda value: re.sub(r"^\s*\d+\.\s*", "", str(value)).strip().casefold()
        exact_district_rows = wide_crop_rows[
            wide_crop_rows["district_name"].map(normalize_location) == normalize_location(selected_district)
        ]
        if not exact_district_rows.empty:
            latest_year = exact_district_rows["year_start"].max()
            recent_rows = exact_district_rows[
                exact_district_rows["year_start"] >= latest_year - 2
            ]
            return max(0.0, float(recent_rows["yield_tonnes_per_hectare"].median()) / 2.47105)

    upag_crop = UPAG_CROP_MAP.get(crop)
    if not upag_crop:
        return None

    matching_crop = upag_df[
        upag_df["crop_name"].astype(str).str.casefold() == upag_crop.casefold()
    ].copy()

    if matching_crop.empty:
        return None

    state_rows = matching_crop[
        matching_crop["state_name"].astype(str).str.casefold()
        == selected_state.casefold()
    ]
    if not state_rows.empty:
        matching_crop = state_rows

    district_rows = matching_crop[
        matching_crop["district_name"].astype(str).str.casefold()
        == selected_district.casefold()
    ]
    if not district_rows.empty:
        matching_crop = district_rows

    chosen_state = str(matching_crop["state_name"].mode().iloc[0])
    chosen_district = str(matching_crop["district_name"].mode().iloc[0])
    reference_area = float(matching_crop["area"].median())
    prediction_year = int(upag_df["year_start"].max())
    available_seasons = sorted(upag_df["season"].dropna().astype(str).unique())

    prediction_rows = pd.DataFrame([
        {
            "state_name": chosen_state,
            "district_name": chosen_district,
            "crop_name": upag_crop,
            "season": season,
            "year_start": prediction_year,
            "area": reference_area
        }
        for season in available_seasons
    ])

    predicted_kg_per_hectare = float(
        np.mean(yield_model.predict(prediction_rows))
    )

    # The supplied normalized UPAG long-format file stores yield in kg/hectare.
    # This conversion is valid for its tonne-based crop records; non-tonne crops
    # (e.g. nuts/bales in separate wide-format tables) must not be merged without
    # a unit conversion table.
    return max(0.0, predicted_kg_per_hectare / 1000.0 / 2.47105)

def best_match(value, choices):

    # Normalize both the user's entry and database choices so matching is
    # case-insensitive and tolerant of extra spaces and spelling mistakes.
    normalized_value = " ".join(str(value).casefold().split())
    normalized_choices = [
        " ".join(str(choice).casefold().split())
        for choice in choices
    ]

    match = process.extractOne(
        normalized_value,
        normalized_choices,
        scorer=fuzz.WRatio
    )

    if match:
        return choices[match[2]]

    return choices[0]

        )

def haversine_km(lat1, lon1, lat2, lon2):

    r = 6371

    p1 = math.radians(lat1)
    p2 = math.radians(lat2)

    dp = math.radians(lat2 - lat1)
    dl = math.radians(lon2 - lon1)

    a = (
        math.sin(dp / 2) ** 2
        + math.cos(p1) * math.cos(p2)
        * math.sin(dl / 2) ** 2
    )

    return 2 * r * math.asin(math.sqrt(a))

def expected_seasonal_rainfall(sowing_month, crop_duration_days):
    month_days = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]
    remaining_days = int(crop_duration_days)
    month = int(sowing_month)
    total_rainfall = 0.0

    while remaining_days > 0:
        days_in_month = month_days[month - 1]
        days_in_period = min(remaining_days, days_in_month)
        monthly_average = float(
            monthly_district_rainfall.get(
                month, district_rain["rainfall_mm"].mean()
            )
        )
        total_rainfall += monthly_average * days_in_period / days_in_month
        remaining_days -= days_in_period
        month = month % 12 + 1

    return total_rainfall

def concise_guideline(value, section):
    """Keep guidance useful and brief, while wrapping long lines cleanly."""
    value = str(value).strip()

    if section in {"common_diseases", "common_pests"}:
        formatted = []
        for item in value.splitlines():
            item = item.strip().lstrip("• ")
            if not item:
                continue
            if "Prevention/management:" in item:
                name, details = item.split(":", 1)
                symptoms, management = details.split("Prevention/management:", 1)
                symptom_sentence = re.split(r"(?<=[.!?])\s+", symptoms.strip())[0]
                management_sentence = re.split(r"(?<=[.!?])\s+", management.strip())[0]
                short_item = (
                    f"{name}: {symptom_sentence} "
                    f"Prevention: {management_sentence}"
                )
            else:
                short_item = item
            formatted.append(textwrap.fill(
                short_item, width=88, initial_indent="  • ",
                subsequent_indent="    "
            ) + "\n")
        return "\n".join(formatted)

    # Keep the first two practical sentences for the other guidance sections.
    sentences = re.split(r"(?<=[.!?])\s+", value)
    short_text = " ".join(sentences[:2]).strip()
    return textwrap.fill(short_text, width=88, subsequent_indent="  ")

def market_date_features(target_date):
    return np.array([[
        target_date.toordinal(),
        np.sin(2 * np.pi * target_date.month / 12),
        np.cos(2 * np.pi * target_date.month / 12)
    ]], dtype=float)


def predict_market_price_for_date(target_market, target_district, target_state, target_date):
    """Fit LinearRegression to prior real or synthetic observations for this date.

    Prefer the selected market's own crop history; fall back to district or state
    history only when a narrower history is too short. Synthetic observations
    remain labelled as estimates and are never represented as real mandi quotes.
    """
    scopes = [
        ("market", crop_market_data[crop_market_data["Market"].astype(str).str.casefold() == str(target_market).casefold()]),
        ("district", crop_market_data[crop_market_data["District"].astype(str).str.casefold() == str(target_district).casefold()]),
        ("state", crop_market_data[crop_market_data["State"].astype(str).str.casefold() == str(target_state).casefold()])
    ]

    chosen_history = None
    chosen_scope = None
    forecast_cutoff = pd.Timestamp(target_date)
    for scope_name, scope_rows in scopes:
        candidate_history = scope_rows[
            scope_rows["Arrival_Date"] < forecast_cutoff
        ]
        if not candidate_history.empty:
            if chosen_history is None:
                chosen_history = candidate_history
                chosen_scope = scope_name
            if candidate_history["Arrival_Date"].nunique() >= 3:
                x_values = np.column_stack([
                    candidate_history["arrival_date_ordinal"].to_numpy(dtype=float),
                    candidate_history["arrival_month_sin"].to_numpy(dtype=float),
                    candidate_history["arrival_month_cos"].to_numpy(dtype=float)
                ])
                y_values = candidate_history["Modal_x0020_Price"].to_numpy(dtype=float)
                trend_model = LinearRegression()
                trend_model.fit(x_values, y_values)
                predicted = float(trend_model.predict(market_date_features(target_date))[0])
                if scope_name == "market" and "data_source" in candidate_history.columns:
                    if candidate_history["data_source"].astype(str).str.startswith("synthetic").all():
                        return max(0.01, predicted), "linear_regression_synthetic_virtual_mandi"
                return max(0.01, predicted), f"linear_regression_{scope_name}"

    # A one-day snapshot has no trend. Use only the latest observed real quote
    # from the narrowest available market/district/state scope as a reference.
    if chosen_history is None:
        return np.nan, "unavailable"
    latest_date = chosen_history["Arrival_Date"].max()
    latest_rows = chosen_history[chosen_history["Arrival_Date"] == latest_date]
    if latest_rows.empty:
        return np.nan, "unavailable"
    return max(0.01, float(latest_rows["Modal_x0020_Price"].median())), f"last_observed_{chosen_scope}_reference"
